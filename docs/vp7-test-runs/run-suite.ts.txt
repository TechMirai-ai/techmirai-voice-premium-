function lossy(ms: ChatMessage[]): ChatMessage[] {
  const out: ChatMessage[] = [];
  for (const m of ms) {
    if (m.role === 'assistant' && !m.tool_calls) continue;
    const prev = out[out.length - 1];
    if (m.role === 'user' && prev?.role === 'user') { out[out.length - 1] = { ...prev, content: `${prev.content} ${m.content}` }; continue; }
    out.push(m);
  }
  return out;
}

// ---------------- ledger + fetch wrapper ----------------
const LEDGER = path.join(RUN_DIR, 'ledger.json');
const today = () => new Date().toLocaleDateString('en-CA');
interface Ledger { days: Record<string, { used: number }>; lastHeaders?: Record<string, string | null> }
function readLedger(): Ledger {
  return existsSync(LEDGER) ? (JSON.parse(readFileSync(LEDGER, 'utf8')) as Ledger) : { days: {} };
}
function writeLedger(l: Ledger) { writeFileSync(LEDGER, JSON.stringify(l, null, 2)); }
const usedToday = () => readLedger().days[today()]?.used ?? 0;
const freeToday = () => DAILY_CAP - usedToday();

class BudgetExhausted extends Error {}

const realFetch = globalThis.fetch;
globalThis.fetch = (async (input: Parameters<typeof fetch>[0], init?: Parameters<typeof fetch>[1]) => {
  if (String(input) !== OPENAI_API_URL) return realFetch(input, init);
  const l = readLedger();
  const used = l.days[today()]?.used ?? 0;
  if (used >= DAILY_CAP) throw new BudgetExhausted(`daily cap ${DAILY_CAP} reached`);
  l.days[today()] = { used: used + 1 }; // counted BEFORE the call, so failures count too
  writeLedger(l);
  if (TEMP !== undefined && init && typeof init.body === 'string') init = { ...init, body: JSON.stringify({ ...JSON.parse(init.body), temperature: Number(TEMP) }) };
  const res = await realFetch(input, init);
  const l2 = readLedger();
  l2.lastHeaders = {
    status: String(res.status),
    'x-ratelimit-limit-requests': res.headers.get('x-ratelimit-limit-requests'),
    'x-ratelimit-remaining-requests': res.headers.get('x-ratelimit-remaining-requests'),
    'x-ratelimit-reset-requests': res.headers.get('x-ratelimit-reset-requests'),
  };
  writeLedger(l2);
  return res;
}) as typeof fetch;

// ---------------- Liquid {{"now" | date: ...}} rendering (tester gap) ----------------
function renderNow(prompt: string): string {
  return prompt.replace(/\{\{"now" \| date: "[^"]+", "([^"]+)"\}\}/g, (_m, tz: string) => {
    const p = Object.fromEntries(
      new Intl.DateTimeFormat('en-US', {
        timeZone: tz, weekday: 'long', month: 'long', day: '2-digit', year: 'numeric',
        hour: '2-digit', minute: '2-digit', hourCycle: 'h23',
      }).formatToParts(FAKE_NOW ? new Date(FAKE_NOW) : new Date()).map((x) => [x.type, x.value]),
    );
    return `${p.weekday}, ${p.month} ${p.day}, ${p.year}, ${p.hour}:${p.minute}`;
  });
}

// ---------------- run state + execution ----------------
interface TurnRecord { tag: string; caller: string; lines: string[]; toolCalls: { name: string; args: string; result: string }[]; requests: number }
interface RunState {
  runId: string; member: string; cases: string; renderedNow: string; startedAt: string;
  messages: ChatMessage[]; nextStep: number; ended: boolean; endReason?: string;
  turns: TurnRecord[]; requests: number; status: 'in-progress' | 'done';
}
const statePath = (id: string) => path.join(RUN2_DIR, `s${SAMPLE}-${id}.json`);
const CB_OK = 'The callback request was saved. Staff will call the caller back.';
const CB_FAIL = 'A system problem prevented this from being saved.';

function toolResult(call: ChatToolCall, failCallback: boolean): { content: string; ends: boolean } {
  const n = call.function.name;
  if (n === 'endCall') return { content: 'Call ended.', ends: true };
  if (n.startsWith('handoff_to_')) return { content: '', ends: true };
  if (n === 'request_callback') return { content: failCallback ? CB_FAIL : CB_OK, ends: false };
  return { content: 'Success.', ends: false }; // log_call_topic (async in production: "Success.")
}

async function runOne(def: RunDef): Promise<'done' | 'budget'> {
  let state: RunState;
  const rendered = await loadRenderedMember(CLIENT, def.member);
  const tools = buildToolDefs(rendered);
  const phrases = rendered.assistant.endCallPhrases ?? [];
  if (existsSync(statePath(def.id))) {
    state = JSON.parse(readFileSync(statePath(def.id), 'utf8')) as RunState;
  } else {
    const base = initialMessages(rendered);
    const sys = base[0]!;
    const renderedSys = renderNow(sys.content ?? '');
    const nowLine = renderedSys.match(/Today:\n(.*)\n/)?.[1] ?? '';
    state = {
      runId: def.id, member: def.member, cases: def.cases, renderedNow: nowLine, startedAt: new Date().toISOString(),
      messages: [{ ...sys, content: renderedSys }, base[1]!, ...(def.seed ?? [])], nextStep: 0, ended: false,
      turns: [], requests: 0, status: 'in-progress',
    };
  }
  const callModel = makeOpenAiCaller(loadEnv().OPENAI_API_KEY!);
  const save = () => writeFileSync(statePath(def.id), JSON.stringify(state, null, 2));

  while (state.nextStep < def.steps.length && !state.ended) {
    const step = def.steps[state.nextStep]!;
    const lastAgent = [...state.messages].reverse().find((m) => m.role === 'assistant' && m.content)?.content ?? '';
    if (step.skipIfAgentSays?.test(lastAgent)) { state.nextStep += 1; save(); continue; }
    if (freeToday() < MIN_FREE_FOR_TURN) { save(); return 'budget'; }

    const before = usedToday();
    let messages: ChatMessage[] = [...state.messages, { role: 'user', content: step.caller }];
    const rec: TurnRecord = { tag: step.tag, caller: step.caller, lines: [], toolCalls: [], requests: 0 };
    let ended = false;
    let endReason: string | undefined;
    try {
      for (let round = 0; round < MAX_ROUNDS; round += 1) {
        const reply = await callModel(LOSSY ? lossy(messages) : messages, tools);
        messages = [...messages, reply];
        if (reply.content) rec.lines.push(reply.content);
        const said = (reply.content ?? '').toLowerCase();
        const phrase = phrases.find((p) => said.includes(p.toLowerCase()));
        const calls = reply.tool_calls ?? [];
        if (calls.length === 0) {
          // Platform emulation: Vapi hangs up once the assistant has SAID an endCallPhrase.
          if (phrase) { ended = true; endReason = `endCallPhrase:${phrase}`; }
          break;
        }
        for (const call of calls) {
          const r = toolResult(call, step.failCallback === true);
          rec.toolCalls.push({ name: call.function.name, args: call.function.arguments, result: r.content });
          messages = [...messages, { role: 'tool', tool_call_id: call.id, name: call.function.name, content: r.content }];
          if (r.ends) { ended = true; endReason = call.function.name; }
        }
        if (phrase && !ended) { ended = true; endReason = `endCallPhrase:${phrase}`; }
        if (ended) break;
      }
    } catch (error) {
      rec.requests = usedToday() - before;
      state.requests += rec.requests;
      save(); // last COMPLETE turn is retained; this partial one is dropped
      if (error instanceof BudgetExhausted) return 'budget';
      throw error;
    }
    rec.requests = usedToday() - before;
    state.requests += rec.requests;
    state.messages = messages; state.nextStep += 1; state.turns.push(rec);
    if (ended) { state.ended = true; state.endReason = endReason as string; }
    save();
  }
  state.status = 'done'; save();
  return 'done';
}

// ---------------- commands ----------------
async function sanity(): Promise<void> {
  const res = await fetch(OPENAI_API_URL, {
    method: 'POST',
    headers: { 'Content-Type': 'application/json', Authorization: `Bearer ${loadEnv().OPENAI_API_KEY}` },
    body: JSON.stringify({ model: OPENAI_MODEL, messages: [{ role: 'user', content: 'Reply with the single word: ok' }], max_tokens: 5 }),
  });
  const body = (await res.json()) as { choices?: { message: { content: string } }[]; error?: { message: string }; usage?: unknown };
  console.log(JSON.stringify({ status: res.status, model: OPENAI_MODEL, reply: body.choices?.[0]?.message.content, error: body.error?.message, usage: body.usage,
    limitRequests: res.headers.get('x-ratelimit-limit-requests'), remainingRequests: res.headers.get('x-ratelimit-remaining-requests'),
    resetRequests: res.headers.get('x-ratelimit-reset-requests'), ledgerUsedToday: usedToday() }, null, 2));
}

    const m = initialMessages(r);
    writeFileSync(path.join(path.dirname(new URL(import.meta.url).pathname), `prompt-${member}.txt`), renderNow(m[0]!.content ?? ''));
    console.log(`--- ${member} firstMessage ---\n${r.assistant.firstMessage}\ntools: ${buildToolDefs(r).map((t) => t.function.name).join(', ')}`);
  }
}

async function main(): Promise<void> {
  mkdirSync(RUN2_DIR, { recursive: true });
  const argv = process.argv.slice(2);
  const si = argv.indexOf('--sample');
  if (si >= 0) { SAMPLE = Number(argv[si + 1]); argv.splice(si, 2); }
  const [cmd, ...rawIds] = argv;
  const ids = rawIds.flatMap((t) => t === '@ja' ? RUNS.filter((r) => r.id.startsWith('ja-') || r.id.startsWith('E-')).map((r) => r.id) : t === '@en' ? RUNS.filter((r) => r.id.startsWith('en-')).map((r) => r.id) : [t]);
  if (cmd === 'sanity') return sanity();
  if (cmd === 'dump') return dump();
  if (cmd === 'run') {
    console.log(`ledger: used ${usedToday()} / cap ${DAILY_CAP} today (${today()})`);
    for (const id of ids) {
      const def = RUNS.find((r) => r.id === id);
      if (!def) { console.log(`unknown run ${id}`); continue; }
      const before = usedToday();
      const result = await runOne(def);
      console.log(`${id}: ${result} (+${usedToday() - before} requests; used ${usedToday()}/${DAILY_CAP})`);
      if (result === 'budget') { console.log('STOP: daily budget reached — resume tomorrow.'); return; }
    }
    return;
  }
  console.log('usage: sanity | dump | run <runId...>');
}
main().catch((e: unknown) => { console.error(e instanceof Error ? e.message : String(e)); process.exitCode = 1; });
